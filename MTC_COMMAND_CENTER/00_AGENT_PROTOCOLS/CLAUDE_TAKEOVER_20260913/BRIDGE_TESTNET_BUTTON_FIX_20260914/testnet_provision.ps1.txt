# TESTNET secret provisioning button - KVM2-P4-03 (owner-only; TESTNET only).
# What it does: copies the provisioning wizard to the Bridge host over SSH and
# opens an interactive SSH session that runs it with sudo. YOU type the two
# values when the wizard asks (hidden input). Nothing is stored on this PC,
# nothing is printed, no service is restarted, nothing is armed.
# Host contact: ONLY the pinned host below (same pin as the dashboard launcher).
$ErrorActionPreference = 'Stop'
$SshExe     = 'C:\Windows\System32\OpenSSH\ssh.exe'
$ScpExe     = 'C:\Windows\System32\OpenSSH\scp.exe'
$HostAddr   = '152.239.123.231'          # srv1856225 (Hostinger KVM2)
$HostUser   = 'baris'
$KeyFile    = Join-Path $env:USERPROFILE '.ssh\hostinger_kvm2'
$KnownHosts = Join-Path $env:USERPROFILE '.ssh\known_hosts'
$Wizard     = Join-Path (Split-Path -Parent $PSScriptRoot) 'tools\KVM2_TESTNET\testnet_provision_wizard.sh'
$Remote     = '/tmp/testnet_provision_wizard.sh'

Write-Host ''
Write-Host '  TESTNET secret provisioning (KVM2-P4-03)' -ForegroundColor Cyan
Write-Host '  Gate: owner-only. Testnet = paper money. HL_LIVE_ACK must never be set.' -ForegroundColor DarkGray
Write-Host ''
foreach ($p in @($SshExe, $ScpExe, $KeyFile, $KnownHosts, $Wizard)) {
    if (-not (Test-Path $p)) { Write-Host "  MISSING: $p" -ForegroundColor Red; Read-Host '  Press Enter to close'; exit 1 }
}
$sshOpts = @('-o', 'StrictHostKeyChecking=yes', '-o', "UserKnownHostsFile=$KnownHosts", '-o', 'ConnectTimeout=20', '-i', $KeyFile)

Write-Host "  1/3 copying the wizard to $HostUser@${HostAddr}:$Remote ..."
& $ScpExe @sshOpts $Wizard "$HostUser@${HostAddr}:$Remote"
if ($LASTEXITCODE -ne 0) { Write-Host '  scp failed (host key, network or key problem). Nothing changed.' -ForegroundColor Red; Read-Host '  Press Enter to close'; exit 1 }

Write-Host '  2/3 opening the wizard on the host. Follow its screens; values are typed there, hidden.'
Write-Host ''
& $SshExe -t @sshOpts "$HostUser@$HostAddr" "sed -i 's/\r$//' $Remote && sudo bash $Remote; rc=`$?; rm -f $Remote; exit `$rc"
$rc = $LASTEXITCODE
# Best-effort cleanup of the remote copy if the interactive session was cut before its own rm ran (the copy holds no secrets).
# Under $ErrorActionPreference='Stop' a native command's redirected stderr line would throw and end the script
# before its final message, so the cleanup runs with 'Continue' and any failure is ignored on purpose.
$ErrorActionPreference = 'Continue'
try { & $SshExe @sshOpts -o BatchMode=yes "$HostUser@$HostAddr" "rm -f $Remote" 2>$null | Out-Null } catch { }
$ErrorActionPreference = 'Stop'

Write-Host ''
if ($rc -eq 0) { Write-Host '  3/3 wizard finished: PROVISIONED. Tell Claude "provisioned".' -ForegroundColor Green }
else { Write-Host "  3/3 wizard ended without PROVISIONED (exit $rc). Nothing else was changed; re-run when ready." -ForegroundColor Yellow }
Read-Host '  Press Enter to close'
